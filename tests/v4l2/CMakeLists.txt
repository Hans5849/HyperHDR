cmake_minimum_required(VERSION 3.16)
project(V4L2CaptureTests LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "V4L2 tests require Linux headers, but no capture hardware")
endif()
enable_testing()
add_executable(v4l2-capture-test main.cpp)
target_include_directories(v4l2-capture-test PRIVATE ../../include)
target_compile_features(v4l2-capture-test PRIVATE cxx_std_17)
target_compile_options(v4l2-capture-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME v4l2-capture COMMAND v4l2-capture-test)
