cmake_minimum_required(VERSION 3.16)
project(LedOutputTests LANGUAGES CXX)
# LED output counters and DDP packetization (issue #9): no Qt, device or network.
enable_testing()
find_package(Threads REQUIRED)
add_executable(led-output-test main.cpp)
target_include_directories(led-output-test PRIVATE ../../include)
target_compile_features(led-output-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(led-output-test PRIVATE /W4 /WX)
else()
  target_compile_options(led-output-test PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(led-output-test PRIVATE Threads::Threads)
add_test(NAME led-output COMMAND led-output-test)
